cmake_minimum_required(VERSION 3.16)
project(TerraMindNavigationTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT ../.. ABSOLUTE)
set(NAV ${ROOT}/TerraMind_FW/Navigation)
set(UPSTREAM ${ROOT}/KF-GINS)
add_library(terramind_navigation STATIC
    ${NAV}/navigation.cpp ${NAV}/core/kf-gins/gi_engine.cpp ${NAV}/core/kf-gins/insmech.cpp)
target_include_directories(terramind_navigation PUBLIC ${NAV} PRIVATE ${NAV}/core ${UPSTREAM}/ThirdParty/eigen-3.3.9)
add_executable(nav_selftest selftest.cpp c_api_check.c)
target_link_libraries(nav_selftest PRIVATE terramind_navigation)
add_executable(nav_reference reference.cpp ${UPSTREAM}/src/kf-gins/gi_engine.cpp ${UPSTREAM}/src/kf-gins/insmech.cpp)
target_include_directories(nav_reference PRIVATE ${NAV} ${UPSTREAM}/src ${UPSTREAM}/ThirdParty/eigen-3.3.9)
# Existing firmware checks: keep DMA log queuing and hardware timestamp capture
# covered without opening any UART or invoking the real application task.
add_executable(monitor_dma_regression ../monitor_dma_queue_test.cpp ${ROOT}/TerraMind_FW/BSP/monitor_uart6_dma.cpp)
target_include_directories(monitor_dma_regression PRIVATE ${ROOT}/tests/mock)
add_executable(time_sync_regression ../time_sync_capture_test.c ${ROOT}/TerraMind_FW/BSP/time_sync_capture.c)
target_include_directories(time_sync_regression PRIVATE ${ROOT}/tests/mock)
add_executable(monitor_log_regression ../monitor_log_test.cpp
    ${ROOT}/TerraMind_FW/App/monitor_app.cpp ${ROOT}/TerraMind_FW/BSP/monitor_uart6_dma.cpp)
target_include_directories(monitor_log_regression PRIVATE ${ROOT}/tests/mock)
add_executable(hal_timebase_regression ../hal_timebase_test.c ${ROOT}/Core/Src/stm32h7xx_hal_timebase_tim.c)
target_include_directories(hal_timebase_regression PRIVATE ${ROOT}/tests/timebase_mock)
add_executable(g365_regression ../g365_capture_test.cpp ${ROOT}/TerraMind_FW/Driver/g365_imu.cpp)
add_executable(gyro_calibration_test ../gyro_calibration_test.cpp ${ROOT}/TerraMind_FW/App/gyro_bias_calibration.cpp)
add_executable(imu_startup_gate_test ../imu_startup_gate_test.cpp
    ${ROOT}/TerraMind_FW/App/app_main.cpp ${ROOT}/TerraMind_FW/App/gyro_bias_calibration.cpp
    ${ROOT}/TerraMind_FW/App/monitor_app.cpp ${ROOT}/TerraMind_FW/Driver/g365_imu.cpp
    ${ROOT}/TerraMind_FW/BSP/monitor_uart6_dma.cpp)
target_include_directories(imu_startup_gate_test PRIVATE ${ROOT}/tests/mock)
foreach(target terramind_navigation nav_selftest nav_reference monitor_dma_regression time_sync_regression monitor_log_regression hal_timebase_regression g365_regression gyro_calibration_test imu_startup_gate_test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /O2 /utf-8 /UNDEBUG)
        target_compile_definitions(${target} PRIVATE _USE_MATH_DEFINES)
    else()
        target_compile_options(${target} PRIVATE -O2 -UNDEBUG)
    endif()
endforeach()
enable_testing()
add_test(NAME upstream_reference COMMAND nav_reference ${CMAKE_BINARY_DIR} ${UPSTREAM}/dataset)
set_tests_properties(upstream_reference PROPERTIES FIXTURES_SETUP reference_data)
add_test(NAME navigation_equivalence_and_runtime COMMAND nav_selftest ${CMAKE_BINARY_DIR} ${UPSTREAM}/dataset)
set_tests_properties(navigation_equivalence_and_runtime PROPERTIES FIXTURES_REQUIRED reference_data)
add_test(NAME existing_monitor_dma COMMAND monitor_dma_regression)
add_test(NAME existing_time_sync COMMAND time_sync_regression)
add_test(NAME existing_monitor_log COMMAND monitor_log_regression)
add_test(NAME hal_millisecond_timebase COMMAND hal_timebase_regression)
add_test(NAME g365_raw_and_delta COMMAND g365_regression)
add_test(NAME gyro_static_calibration COMMAND gyro_calibration_test)
add_test(NAME silent_startup_and_raw_logging COMMAND imu_startup_gate_test)
