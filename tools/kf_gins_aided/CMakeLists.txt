cmake_minimum_required(VERSION 3.16)
project(TerraMindOfflineAiding LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT ../.. ABSOLUTE)
set(CORE ${ROOT}/TerraMind_FW/Navigation/core)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/KF-GINS/ThirdParty/yaml-cpp-0.7.0 yaml-cpp EXCLUDE_FROM_ALL)
add_library(aided_core STATIC ${CORE}/kf-gins/gi_engine.cpp ${CORE}/kf-gins/insmech.cpp)
target_include_directories(aided_core PUBLIC ${CORE} ${ROOT}/KF-GINS/ThirdParty/eigen-3.3.9)
add_executable(kf_gins_aided main.cpp)
target_link_libraries(kf_gins_aided PRIVATE aided_core yaml-cpp)
add_executable(aiding_test ${ROOT}/tests/kf_gins_aiding_test.cpp)
target_include_directories(aiding_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(aiding_test PRIVATE aided_core)
foreach(target aided_core kf_gins_aided aiding_test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /O2 /utf-8 /UNDEBUG)
        target_compile_definitions(${target} PRIVATE _USE_MATH_DEFINES)
    else()
        target_compile_options(${target} PRIVATE -O2 -UNDEBUG)
    endif()
endforeach()
enable_testing()
add_test(NAME aiding_models_and_updates COMMAND aiding_test)
